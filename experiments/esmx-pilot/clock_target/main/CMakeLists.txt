get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(PILOT "${ROOT}/main/src/Audio/EsMxPilot")
set(KWS "${ROOT}/main/src/Audio/EsMxKws")
set(BANK "${ROOT}/experiments/esmx-compactas")
set(PHONEME "${ROOT}/experiments/esmx-pilot/phoneme_native")

idf_component_register(
    SRCS "main.cpp"
        "${PILOT}/ClockAcoustic.cpp" "${PILOT}/ClockModelVfs.cpp"
        "${PILOT}/Recognition.cpp" "${PILOT}/JokeFallback.cpp"
        "${KWS}/Compact.cpp" "${KWS}/logmel.cpp"
        "${BANK}/guard/PhraseGate.cpp" "${BANK}/autonomous/WordDice.cpp"
        "${BANK}/autonomous/AcousticDice.cpp"
        "${PHONEME}/Kernel.cpp" "${PHONEME}/Alignment.cpp"
        "${PHONEME}/JokeGuard.cpp" "${PHONEME}/ClockGuard.cpp"
    REQUIRES esp_timer vfs mbedtls
    INCLUDE_DIRS "${ROOT}/main/src" "${KWS}" "${PHONEME}" "${BANK}/guard" "${BANK}/autonomous"
        "${BB_ESMX_ASSETS}" "${BB_ESMX_SPHINX_INCLUDE}" "${BB_ESMX_SPHINX_GENERATED_INCLUDE}"
        "${BB_ESMX_CLOCK_PACK}" "${BB_ESMX_CLOCK_TARGET_CASES}")

add_prebuilt_library(bb_esmx_target_sphinx "${BB_ESMX_SPHINX_LIBRARY}")
target_link_libraries(${COMPONENT_LIB} PRIVATE bb_esmx_target_sphinx)
target_link_options(${COMPONENT_LIB} INTERFACE "-Wl,-u,__wrap_longjmp")
target_compile_definitions(${COMPONENT_LIB} PRIVATE
                           BB_ESMX_PILOT BB_ESMX_JOKE_FALLBACK BB_ESMX_CLOCK_FALLBACK BB_ESMX_CLOCK_ACOUSTIC)
target_compile_options(${COMPONENT_LIB} PRIVATE -ffp-contract=off)

option(BB_ESMX_BUTTER_ACOUSTIC "Include the experimental Mantequilla validation cases" OFF)
option(BB_ESMX_BUTTER_WEAK_CONFLICT "Allow the experimental weak D6 conflict for Mantequilla" OFF)
if(BB_ESMX_BUTTER_ACOUSTIC)
    if(NOT IS_ABSOLUTE "${BB_ESMX_BUTTER_ASSETS}" OR
       NOT EXISTS "${BB_ESMX_BUTTER_ASSETS}/ButterModelAssets.h")
        message(FATAL_ERROR "Set BB_ESMX_BUTTER_ASSETS to the local Butter validation assets")
    endif()
    target_compile_definitions(${COMPONENT_LIB} PRIVATE BB_ESMX_BUTTER_ACOUSTIC BB_ESMX_BUTTER_BENCHMARK)
    target_include_directories(${COMPONENT_LIB} PRIVATE "${BB_ESMX_BUTTER_ASSETS}")
    target_sources(${COMPONENT_LIB} PRIVATE "${PILOT}/ButterAcoustic.cpp")
    if(BB_ESMX_BUTTER_WEAK_CONFLICT)
        target_compile_definitions(${COMPONENT_LIB} PRIVATE BB_ESMX_BUTTER_WEAK_CONFLICT)
    endif()
elseif(BB_ESMX_BUTTER_WEAK_CONFLICT)
    message(FATAL_ERROR "BB_ESMX_BUTTER_WEAK_CONFLICT requires BB_ESMX_BUTTER_ACOUSTIC")
endif()

option(BB_ESMX_D6_SCOPE_BENCH "Measure the experimental D6 scope in the isolated ESP32-S3 benchmark" OFF)
option(BB_ESMX_D6_BENCH_TRACE "Print D6 intermediate evidence during the isolated benchmark" OFF)
option(BB_ESMX_D6_BENCH_OMIT_KWS "Check D6 missing-asset behavior in the isolated benchmark" OFF)
if(BB_ESMX_D6_BENCH_OMIT_KWS AND NOT BB_ESMX_D6_SCOPE_BENCH)
    message(FATAL_ERROR "D6 missing-asset test requires the isolated D6 benchmark")
endif()
if(BB_ESMX_D6_SCOPE_BENCH)
    set(DICE_SCOPE "${ROOT}/experiments/esmx-pilot/dice_scope")
    set(DICE_SCOPE_KWS "${DICE_SCOPE}/dice-scope.kws")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${DICE_SCOPE_KWS}")
    file(SHA256 "${DICE_SCOPE_KWS}" DICE_SCOPE_KWS_SHA)
    if(NOT DICE_SCOPE_KWS_SHA STREQUAL "2901388198b5b468b4bc6bdbc4d5588aee8ae7100878d80c78edf058f146dff0")
        message(FATAL_ERROR "D6 benchmark keyword file differs from the reviewed SHA: ${DICE_SCOPE_KWS_SHA}")
    endif()
    file(READ "${DICE_SCOPE_KWS}" DICE_SCOPE_KWS_HEX HEX)
    string(REGEX REPLACE "(..)" "0x\\1," DICE_SCOPE_KWS_BYTES "${DICE_SCOPE_KWS_HEX}")
    configure_file(DiceScopeKws.h.in DiceScopeKws.h @ONLY)
    target_include_directories(${COMPONENT_LIB} PRIVATE "${DICE_SCOPE}" "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE BB_ESMX_D6_SCOPE_BENCH)
    if(BB_ESMX_D6_BENCH_TRACE)
        target_compile_definitions(${COMPONENT_LIB} PRIVATE BB_ESMX_D6_SCOPE_TRACE)
    endif()
    if(BB_ESMX_D6_BENCH_OMIT_KWS)
        target_compile_definitions(${COMPONENT_LIB} PRIVATE BB_ESMX_D6_BENCH_OMIT_KWS)
    endif()
    target_sources(${COMPONENT_LIB} PRIVATE "${DICE_SCOPE}/DiceScope.cpp")
endif()

option(BB_ESMX_BUTTER_CEPSTRA "Reuse Mantequilla cepstra in the isolated benchmark" OFF)
option(BB_ESMX_EXPECT_OPPOSING_POOL "Verify the pinned opposing FSG pool in the isolated benchmark" OFF)
option(BB_ESMX_BUTTER_APP_EXPERIMENT "Replay the Mantequilla-only application profile" OFF)
set(BB_ESMX_BUTTER_APP_FAIL_STAGE "0" CACHE STRING "Inject a replay-only Butter install error (1 KWS, 2 affirm, 3 oppose)")
if(NOT BB_ESMX_BUTTER_APP_FAIL_STAGE MATCHES "^[0123]$")
    message(FATAL_ERROR "BB_ESMX_BUTTER_APP_FAIL_STAGE must be 0, 1, 2 or 3")
endif()
if(BB_ESMX_BUTTER_APP_EXPERIMENT)
    if(NOT BB_ESMX_BUTTER_ACOUSTIC OR BB_ESMX_D6_SCOPE_BENCH OR BB_ESMX_BUTTER_CEPSTRA OR BB_ESMX_BUTTER_WEAK_CONFLICT)
        message(FATAL_ERROR "For application replay enable acoustic Mantequilla alone. The profile selects its own cepstra and conflict settings.")
    endif()
    include("${ROOT}/experiments/esmx-pilot/butter_acoustic/app_profile.cmake")
endif()
if(NOT BB_ESMX_BUTTER_APP_FAIL_STAGE STREQUAL "0")
    if(NOT BB_ESMX_BUTTER_APP_EXPERIMENT OR NOT BB_ESMX_EXPECT_OPPOSING_POOL)
        message(FATAL_ERROR "Butter fault injection requires application replay and pool verification")
    endif()
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        BB_ESMX_BUTTER_APP_FAIL_STAGE=${BB_ESMX_BUTTER_APP_FAIL_STAGE})
    target_link_options(${COMPONENT_LIB} INTERFACE "LINKER:--wrap=ps_add_kws,--wrap=ps_add_jsgf_string")
endif()
if(BB_ESMX_BUTTER_CEPSTRA OR BB_ESMX_EXPECT_OPPOSING_POOL)
    if(NOT BB_ESMX_BUTTER_ACOUSTIC)
        message(FATAL_ERROR "The isolated Butter performance options require BB_ESMX_BUTTER_ACOUSTIC")
    endif()
endif()
if(BB_ESMX_BUTTER_CEPSTRA)
    if(NOT EXISTS "${BB_ESMX_SPHINX_PRIVATE_INCLUDE}/pocketsphinx_internal.h")
        message(FATAL_ERROR "Set BB_ESMX_SPHINX_PRIVATE_INCLUDE to the exact source used by libpocketsphinx")
    endif()
    set(BRIDGE "${ROOT}/experiments/esmx-pilot/butter_acoustic")
    target_include_directories(${COMPONENT_LIB} PRIVATE "${BRIDGE}" "${BB_ESMX_SPHINX_PRIVATE_INCLUDE}")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE BB_ESMX_BUTTER_CEPSTRA)
    target_sources(${COMPONENT_LIB} PRIVATE "${BRIDGE}/cepstra_bridge.c")
endif()
if(BB_ESMX_EXPECT_OPPOSING_POOL)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE BB_ESMX_EXPECT_OPPOSING_POOL)
endif()
