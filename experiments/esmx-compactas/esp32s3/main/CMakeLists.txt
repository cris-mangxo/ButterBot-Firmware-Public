if(TRANSCRIPT_GRAMMAR_BENCH AND (NOT KWS_WITH_MODEL OR NOT PHRASE_BANK_DIR))
    message(FATAL_ERROR "Transcript grammar bench requires the acoustic phrase bank")
endif()
if(WORD_DICE_DIR AND (NOT KWS_WITH_MODEL OR NOT PHRASE_BANK_DIR OR TRANSCRIPT_GRAMMAR_BENCH))
    message(FATAL_ERROR "Word dice bench requires the phrase bank and no transcript grammar")
endif()
if(WORD_DICE_TEST_PCM_DIR AND NOT WORD_DICE_DIR)
    message(FATAL_ERROR "PCM test requires the word dice bank")
endif()
if(WORD_DICE_EXTRA_DIR AND NOT WORD_DICE_DIR)
    message(FATAL_ERROR "Supplemental word bank requires the original word dice bank")
endif()

if(WORD_DICE_ADAPTED_DIR AND NOT WORD_DICE_EXTRA_DIR)
    message(FATAL_ERROR "Human-adapted bank requires the previously evaluated supplemental bank")
endif()

if(CONTINUOUS_DICE_DIR AND NOT WORD_DICE_ADAPTED_DIR)
    message(FATAL_ERROR "Continuous dice bench requires the earlier word banks")
endif()

if(CONTINUOUS_DICE_GUARD AND NOT CONTINUOUS_DICE_DIR)
    message(FATAL_ERROR "Continuous dice guard requires the continuous bank")
endif()

if(CONTINUOUS_DICE_RESCUE AND NOT CONTINUOUS_DICE_GUARD)
    message(FATAL_ERROR "Ordered dice rescue requires the continuous guard")
endif()

set(KWS_SOURCE "../../../../main/src/Audio/EsMxKws")
if(KWS_WITH_MODEL)
    if(NOT EXISTS "${PILOT_MODEL_DIR}/pilot_model.h")
        message(FATAL_ERROR "Generate the synthetic pilot_model.h before building")
    endif()
    idf_component_register(SRCS "main.cpp" "${KWS_SOURCE}/logmel.cpp" "${KWS_SOURCE}/Compact.cpp"
                           INCLUDE_DIRS "${KWS_SOURCE}" "${PILOT_MODEL_DIR}")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE KWS_WITH_MODEL)
    if(DICE6_ANCHOR_DIR AND PHRASE_BANK_DIR)
        message(FATAL_ERROR "Select one experimental gate for the bench")
    elseif(DICE6_ANCHOR_DIR)
        if(NOT EXISTS "${DICE6_ANCHOR_DIR}/dice6_anchors.h")
            message(FATAL_ERROR "Generate the synthetic dice6_anchors.h before building")
        endif()
        target_sources(${COMPONENT_LIB} PRIVATE "../../guard/Dice6Guard.cpp")
        target_include_directories(${COMPONENT_LIB} PRIVATE "../.." "../../guard"
                                   "../../../../main/src" "${DICE6_ANCHOR_DIR}")
        target_compile_definitions(${COMPONENT_LIB} PRIVATE KWS_WITH_DICE6_ANCHORS)
    elseif(PHRASE_BANK_DIR)
        if(NOT EXISTS "${PHRASE_BANK_DIR}/phrase_bank.h")
            message(FATAL_ERROR "Generate the synthetic phrase_bank.h before building")
        endif()
        target_sources(${COMPONENT_LIB} PRIVATE "../../guard/PhraseGate.cpp")
        target_include_directories(${COMPONENT_LIB} PRIVATE "../../guard"
                                   "../../../../main/src" "${PHRASE_BANK_DIR}")
        target_compile_definitions(${COMPONENT_LIB} PRIVATE KWS_WITH_PHRASE_BANK)
        if(TRANSCRIPT_GRAMMAR_BENCH)
            target_sources(${COMPONENT_LIB} PRIVATE "../../transcript/DiceGrammar.cpp")
            target_include_directories(${COMPONENT_LIB} PRIVATE "../../transcript")
            target_compile_definitions(${COMPONENT_LIB} PRIVATE KWS_WITH_TRANSCRIPT_GRAMMAR)
        elseif(WORD_DICE_DIR)
            if(NOT EXISTS "${WORD_DICE_DIR}/word_bank.h")
                message(FATAL_ERROR "Generate word_bank.h before building")
            endif()
            target_sources(${COMPONENT_LIB} PRIVATE "../../autonomous/WordDice.cpp")
            target_include_directories(${COMPONENT_LIB} PRIVATE "../../autonomous" "${WORD_DICE_DIR}")
            target_compile_definitions(${COMPONENT_LIB} PRIVATE KWS_WITH_WORD_DICE)
            if(WORD_DICE_EXTRA_DIR)
                if(NOT EXISTS "${WORD_DICE_EXTRA_DIR}/word_extra.h")
                    message(FATAL_ERROR "Generate the supplemental word_extra.h before building")
                endif()
                target_include_directories(${COMPONENT_LIB} PRIVATE "${WORD_DICE_EXTRA_DIR}")
                target_compile_definitions(${COMPONENT_LIB} PRIVATE KWS_WITH_WORD_DICE_EXTRA)
                if(WORD_DICE_ADAPTED_DIR)
                    if(NOT EXISTS "${WORD_DICE_ADAPTED_DIR}/word_adapted.h")
                        message(FATAL_ERROR "Generate the human-adapted word_adapted.h before building")
                    endif()
                    target_include_directories(${COMPONENT_LIB} PRIVATE "${WORD_DICE_ADAPTED_DIR}")
                    target_compile_definitions(${COMPONENT_LIB} PRIVATE KWS_WITH_WORD_DICE_ADAPTED)
                    if(CONTINUOUS_DICE_DIR)
                        if(NOT EXISTS "${CONTINUOUS_DICE_DIR}/acoustic_dice.h")
                            message(FATAL_ERROR "Generate the continuous acoustic_dice.h before building")
                        endif()
                        target_sources(${COMPONENT_LIB} PRIVATE "../../autonomous/AcousticDice.cpp")
                        target_include_directories(${COMPONENT_LIB} PRIVATE "${CONTINUOUS_DICE_DIR}")
                        target_compile_definitions(${COMPONENT_LIB} PRIVATE KWS_WITH_CONTINUOUS_DICE)
                        if(CONTINUOUS_DICE_GUARD)
                            target_compile_definitions(${COMPONENT_LIB} PRIVATE KWS_WITH_CONTINUOUS_DICE_GUARD)
                            if(CONTINUOUS_DICE_RESCUE)
                                target_compile_definitions(${COMPONENT_LIB} PRIVATE KWS_WITH_CONTINUOUS_DICE_RESCUE)
                            endif()
                        endif()
                    endif()
                endif()
            endif()
            if(WORD_DICE_TEST_PCM_DIR)
                if(NOT EXISTS "${WORD_DICE_TEST_PCM_DIR}/word_pcm.h")
                    message(FATAL_ERROR "Generate the PCM cases before building")
                endif()
                target_include_directories(${COMPONENT_LIB} BEFORE PRIVATE "${WORD_DICE_TEST_PCM_DIR}")
                target_compile_definitions(${COMPONENT_LIB} PRIVATE KWS_WITH_WORD_TEST_PCM)
            endif()
        endif()
    endif()
else()
    idf_component_register(SRCS "main.cpp")
endif()
